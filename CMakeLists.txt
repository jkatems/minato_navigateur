cmake_minimum_required(VERSION 3.21)
project(Minato VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets WebEngineWidgets Network Sql)
add_library(minato_core STATIC src/database/Store.cpp src/browser/UrlResolver.cpp src/reports/DiagnosticReport.cpp src/reports/SmtpClient.cpp)
target_include_directories(minato_core PUBLIC src)
target_link_libraries(minato_core PUBLIC Qt6::Widgets Qt6::Sql Qt6::Network)
add_library(minato_app STATIC src/profiles/Profile.cpp src/sync/SyncClient.cpp src/sync/SyncPage.cpp src/reports/ReportDialog.cpp src/network/NetworkManager.cpp src/network/InterfaceType.cpp src/browser/BrowserTab.cpp src/ui/InternalPages.cpp src/ui/MainWindow.cpp src/ui/Theme.cpp resources/minato.qrc src/downloads/DownloadManager.cpp)
target_link_libraries(minato_app PUBLIC minato_core Qt6::WebEngineWidgets)
qt_add_executable(minato src/main.cpp)
set_target_properties(minato PROPERTIES WIN32_EXECUTABLE TRUE)
target_link_libraries(minato PRIVATE minato_app)
foreach(target minato minato_core minato_app)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /utf-8)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
 endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 qt_add_executable(core_tests tests/CoreTests.cpp)
 target_link_libraries(core_tests PRIVATE minato_core Qt6::Test)
 add_test(NAME core COMMAND core_tests -platform offscreen)
 qt_add_executable(report_tests tests/ReportTests.cpp)
 target_link_libraries(report_tests PRIVATE minato_core Qt6::Test)
 add_test(NAME reports COMMAND report_tests)
 qt_add_executable(smtp_tests tests/SmtpTests.cpp)
 target_link_libraries(smtp_tests PRIVATE minato_core Qt6::Test)
 target_compile_definitions(smtp_tests PRIVATE SMTP_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
 add_test(NAME smtp COMMAND smtp_tests)
 set_tests_properties(smtp PROPERTIES TIMEOUT 60)
 qt_add_executable(sync_tests tests/SyncTests.cpp)
 target_link_libraries(sync_tests PRIVATE minato_app Qt6::Test)
 add_test(NAME sync COMMAND sync_tests)
 set_tests_properties(sync PROPERTIES TIMEOUT 60)
 qt_add_executable(browser_tests tests/BrowserTests.cpp)
 target_link_libraries(browser_tests PRIVATE minato_app Qt6::Test)
 add_test(NAME browser COMMAND browser_tests)
 set_tests_properties(browser PROPERTIES TIMEOUT 90)
endif()
install(TARGETS minato RUNTIME DESTINATION bin)
if(UNIX AND NOT APPLE)
 install(FILES packaging/minato.desktop DESTINATION share/applications)
 install(FILES resources/icons/minato.svg DESTINATION share/icons/hicolor/scalable/apps)
endif()
if(WIN32)
 # App-local MSVC runtime: the ZIP does not require installing vc_redist first.
 set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
 include(InstallRequiredSystemLibraries)
 install(FILES launch-exam.cmd DESTINATION .)
 install(FILES packaging/WINDOWS-README.txt DESTINATION . RENAME LISEZ-MOI.txt)
 qt_generate_deploy_app_script(TARGET minato OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
 install(SCRIPT ${deploy_script})
endif()

set(CPACK_PACKAGE_NAME "minato")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Minato desktop browser")
set(CPACK_PACKAGE_VENDOR "Minato")
set(CPACK_GENERATOR "ZIP")
if(WIN32)
 set(CPACK_PACKAGE_FILE_NAME "Minato-${PROJECT_VERSION}-windows-x64")
 set(CPACK_PACKAGE_INSTALL_DIRECTORY "Minato")
 set(CPACK_NSIS_DISPLAY_NAME "Minato")
 set(CPACK_NSIS_PACKAGE_NAME "Minato")
 set(CPACK_NSIS_EXECUTABLES_DIRECTORY "bin")
 set(CPACK_PACKAGE_EXECUTABLES "minato" "Minato")
 set(CPACK_NSIS_MODIFY_PATH OFF)
 set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
endif()
include(CPack)
